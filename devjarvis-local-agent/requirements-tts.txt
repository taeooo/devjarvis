# Install this file only when high-quality local voice output is needed.
#
# MeloTTS-Korean is the recommended first provider:
#   DEVJARVIS_LOCAL_AGENT_TTS_PROVIDER=melotts_kr
#
# Piper remains supported for users who already have a Piper voice model:
#   DEVJARVIS_LOCAL_AGENT_TTS_PROVIDER=piper_cli
